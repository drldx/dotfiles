Check the version in official github repo named downloads

Look for the assest session of the required version
Download the x86 linux tar.gz file 

Follow the extraction code form the official site :

  sudo rm -rf /opt/nvim-linux-x86_64
  sudo tar -C /opt -xzf nvim-linux-x86_64.tar.gz

CleanUp;

#Remove Neovim executable/symlink
  sudo rm -f /usr/local/bin/nvim

#Remove manual installs in /opt
  sudo rm -rf /opt/nvim*
  sudo rm -rf /usr/local/nvim*

#Remove AppImage copies if any
  rm -f ~/.local/bin/nvim*
  rm -f ~/Downloads/nvim*

#Remove Neovim config
  rm -rf ~/.config/nvim

#Remove plugins and installed packages
  rm -rf ~/.local/share/nvim

#Remove cache
  rm -rf ~/.cache/nvim

#Remove state/session/shada files
  rm -rf ~/.local/state/nvim

#Remove extra local data
  rm -rf ~/.local/share/nvim-data

#Verify cleanup
  which nvim
